function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": "Keine POST-Daten empfangen"})).setMimeType(ContentService.MimeType.JSON);
    }

    var sheetApp = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    
    var rawDate = data.abholDatum || "Allgemein";
    var sheetName = rawDate.replace(/[/\?%*:|"< >]/g, '_').substring(0, 30);
    
    var sheet = sheetApp.getSheetByName(sheetName);
    var baseHeaders = ["Zeitstempel", "Nachname", "Vorname", "E-Mail", "Abholdatum", "Gesamtgewicht", "Gesamtsumme", "Kommentar"];
    
    // 1. Tabellenblatt anlegen, falls es noch nicht existiert
    if (!sheet) {
      sheet = sheetApp.insertSheet(sheetName);
      var initHeaders = baseHeaders.slice();
      
      var prodsToAdd = [];
      if (data.allProductNames && data.allProductNames.length > 0) {
        prodsToAdd = data.allProductNames;
      } else if (data.items && data.items.length > 0) {
        prodsToAdd = data.items.map(function(it) { return it.name; });
      }
      
      prodsToAdd.forEach(function(pName) {
        if (initHeaders.indexOf(pName) === -1) {
          initHeaders.push(pName);
        }
      });
      
      sheet.appendRow(initHeaders);
      var hRange = sheet.getRange(1, 1, 1, initHeaders.length);
      hRange.setFontWeight("bold").setBackground("#8b0000").setFontColor("#ffffff");
      sheet.setFrozenRows(1);
    }
    
    // 2. Erfasse alle Produkte, die als Spalte existieren müssen
    var neededProducts = [];
    if (data.allProductNames && data.allProductNames.length > 0) {
      neededProducts = data.allProductNames;
    } else if (data.items && data.items.length > 0) {
      neededProducts = data.items.map(function(it) { return it.name; });
    }
    
    // 3. Prüfe vorhandene Kopfzeile in Google Sheets & ergänze fehlende Spalten
    for (var p = 0; p < neededProducts.length; p++) {
      var prodName = neededProducts[p];
      var currentHeaders = sheet.getRange(1, 1, 1, Math.max(1, sheet.getLastColumn())).getValues()[0];
      
      var found = false;
      for (var c = 0; c < currentHeaders.length; c++) {
        if (String(currentHeaders[c]).trim().toLowerCase() === String(prodName).trim().toLowerCase()) {
          found = true;
          break;
        }
      }
      
      if (!found) {
        var nextCol = currentHeaders.length + 1;
        sheet.getRange(1, nextCol).setValue(prodName).setFontWeight("bold").setBackground("#8b0000").setFontColor("#ffffff");
      }
    }
    
    // 4. Frische finale Kopfzeile einlesen
    var finalHeaders = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    var newRow = new Array(finalHeaders.length).fill("");
    
    newRow[0] = new Date().toLocaleString("de-DE");
    newRow[1] = data.nachname || "";
    newRow[2] = data.vorname || "";
    newRow[3] = data.email || "";
    newRow[4] = data.abholDatum || "";
    
    var totalGrammOrder = 0;
    
    // Befülle Produktspalten ausschließlich mit reinen Zahlenwerten
    if (data.items && data.items.length > 0) {
      data.items.forEach(function(item) {
        var colIndex = -1;
        for (var h = 0; h < finalHeaders.length; h++) {
          if (String(finalHeaders[h]).trim().toLowerCase() === String(item.name).trim().toLowerCase()) {
            colIndex = h;
            break;
          }
        }
        if (colIndex !== -1) {
          newRow[colIndex] = Number(item.qty) || 0;
        }
        if (item.totalWeightGramm) {
          totalGrammOrder += item.totalWeightGramm;
        }
      });
    }

    // Spalten-Indizes ermitteln
    var weightColIndex = -1;
    var sumColIndex = -1;
    var kommentarColIndex = -1;

    for (var h = 0; h < finalHeaders.length; h++) {
      var headerLower = String(finalHeaders[h]).toLowerCase();
      if (headerLower.indexOf("gewicht") !== -1 || headerLower.indexOf("fleischbedarf") !== -1) {
        weightColIndex = h;
      } else if (headerLower.indexOf("gesamtsumme") !== -1 || headerLower.indexOf("summe") !== -1) {
        sumColIndex = h;
      } else if (headerLower.indexOf("kommentar") !== -1) {
        kommentarColIndex = h;
      }
    }

    if (weightColIndex !== -1) {
      newRow[weightColIndex] = totalGrammOrder / 1000;
    }

    if (sumColIndex !== -1) {
      newRow[sumColIndex] = (typeof data.totalSum === 'number') ? data.totalSum : 0;
    }

    if (kommentarColIndex !== -1) {
      newRow[kommentarColIndex] = data.kommentar || "";
    }
    
    // Alte GESAMTSUMME-Zeile löschen
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      var lastRowFirstCell = sheet.getRange(lastRow, 1).getValue();
      if (String(lastRowFirstCell).indexOf("GESAMTSUMME") !== -1) {
        sheet.deleteRow(lastRow);
      }
    }
    
    // Neue Datenzeile anhängen
    sheet.appendRow(newRow);
    var addedRowIndex = sheet.getLastRow();

    // Zahlen-Formate anwenden
    if (weightColIndex !== -1) {
      sheet.getRange(addedRowIndex, weightColIndex + 1).setNumberFormat('0.00" kg"');
    }
    if (sumColIndex !== -1) {
      sheet.getRange(addedRowIndex, sumColIndex + 1).setNumberFormat('0.00" €"');
    }
    
    // GESAMTSUMME-Zeile am Tabellenende aufbauen
    var newLastRow = sheet.getLastRow();
    if (newLastRow >= 2) {
      var sumRow = new Array(finalHeaders.length).fill("");
      sumRow[0] = "GESAMTSUMME";
      
      for (var colIdx = 0; colIdx < finalHeaders.length; colIdx++) {
        var hName = String(finalHeaders[colIdx]).trim();
        var hLower = hName.toLowerCase();
        var colLetter = getColLetter(colIdx + 1);
        
        if (hLower.indexOf("gesamtsumme") !== -1 || hLower.indexOf("gewicht") !== -1 || hLower.indexOf("fleischbedarf") !== -1 || (colIdx >= 8 && hLower !== "kommentar")) {
          sumRow[colIdx] = '=SUM(' + colLetter + '2:' + colLetter + newLastRow + ')';
        }
      }
      
      sheet.appendRow(sumRow);
      var sumRowIndex = sheet.getLastRow();

      if (weightColIndex !== -1) {
        sheet.getRange(sumRowIndex, weightColIndex + 1).setNumberFormat('0.00" kg"');
      }
      if (sumColIndex !== -1) {
        sheet.getRange(sumRowIndex, sumColIndex + 1).setNumberFormat('0.00" €"');
      }

      var sumRange = sheet.getRange(sumRowIndex, 1, 1, finalHeaders.length);
      sumRange.setFontWeight("bold").setBackground("#e8f5e9");
    }

    // 5. Bestätigungs-E-Mail senden
    var recipientEmail = (data.emailRaw || data.email || "").trim();
    if (recipientEmail === "Nicht angegeben" || recipientEmail === "Ohne Benachrichtigung") {
      recipientEmail = "";
    }

    var isOptedIn = data.emailBenachrichtigung !== false;

    if (isOptedIn && recipientEmail.indexOf("@") !== -1) {
      try {
        var mailSubject = "Bestellbestätigung - Metzgerei Peter (" + (data.abholDatum || "Abholung") + ")";
        
        var mailBody = "Hallo " + (data.vorname || "") + " " + (data.nachname || "") + ",\n\n" +
          "vielen Dank für deine Bestellung bei der Metzgerei Peter!\n\n" +
          "----------------------------------------\n" +
          "Abholtermin: " + (data.abholDatum || "") + "\n" +
          "----------------------------------------\n\n" +
          "Bestellte Artikel:\n";
          
        if (data.items && data.items.length > 0) {
          data.items.forEach(function(item) {
            var costStr = (Number(item.cost) || 0).toFixed(2).replace('.', ',');
            mailBody += " • " + item.name + ": " + item.displayQty + " (" + costStr + " €)\n";
          });
        } else if (data.zusammenfassung) {
          mailBody += " • " + data.zusammenfassung + "\n";
        }
        
        mailBody += "\nGesamtsumme: " + (data.gesamtsumme || "0,00") + " €\n\n";
        
        if (data.kommentar && data.kommentar !== "Kein Kommentar") {
          mailBody += "Anmerkung / Hinweis:\n" + data.kommentar + "\n\n";
        }
        
        mailBody += "Herzliche Grüße,\nMetzgerei Peter";

        // Versuch 1: MailApp, Versuch 2: GmailApp
        try {
          MailApp.sendEmail({
            to: recipientEmail,
            subject: mailSubject,
            body: mailBody,
            name: "Metzgerei Peter"
          });
        } catch (e1) {
          GmailApp.sendEmail(recipientEmail, mailSubject, mailBody, {
            name: "Metzgerei Peter"
          });
        }
      } catch (mailError) {
        Logger.log("E-Mail Fehler: " + mailError.toString());
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var action = e && e.parameter ? e.parameter.action : null;
    if (action === "getOrders") {
      var sheetApp = SpreadsheetApp.getActiveSpreadsheet();
      var rawDate = e.parameter.abholDatum || "Allgemein";
      var sheetName = rawDate.replace(/[/\?%*:|"< >]/g, '_').substring(0, 30);
      var sheet = sheetApp.getSheetByName(sheetName);
      
      if (!sheet) {
        return ContentService.createTextOutput(JSON.stringify({ "result": "success", "orders": [], "headers": [] })).setMimeType(ContentService.MimeType.JSON);
      }
      
      var data = sheet.getDataRange().getValues();
      if (data.length <= 1) {
        return ContentService.createTextOutput(JSON.stringify({ "result": "success", "orders": [], "headers": [] })).setMimeType(ContentService.MimeType.JSON);
      }
      
      var headers = data[0];
      var orders = [];
      
      for (var i = 1; i < data.length; i++) {
        var row = data[i];
        if (String(row[0]).indexOf("GESAMTSUMME") !== -1) continue;
        
        var orderObj = {};
        for (var j = 0; j < headers.length; j++) {
          orderObj[headers[j]] = row[j];
        }
        orders.push(orderObj);
      }
      
      return ContentService.createTextOutput(JSON.stringify({ "result": "success", "orders": orders, "headers": headers })).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "message": "Unbekannte Aktion" })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "message": err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getColLetter(colIdx) {
  var letter = '';
  while (colIdx > 0) {
    var temp = (colIdx - 1) % 26;
    letter = String.fromCharCode(65 + temp) + letter;
    colIdx = Math.floor((colIdx - 1) / 26);
  }
  return letter;
}
