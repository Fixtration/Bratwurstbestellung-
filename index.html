Google Sheets Script

Automatische Tab-Erstellung pro Abholtermin, Fleischgewicht in kg & Summenzeile

Kopiere diesen Code in deinen Google Apps Script Editor (*Erweiterungen → Apps Script* in Google Sheets):

function doPost(e) {
  try {
    var sheetApp = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    
    var rawDate = data.abholDatum || "Allgemein";
    var sheetName = rawDate.replace(/[/\\?%*:|"< >]/g, '_').substring(0, 30);
    
    var sheet = sheetApp.getSheetByName(sheetName);
    
    if (!sheet) {
      sheet = sheetApp.insertSheet(sheetName);
      var headers = ["Zeitstempel", "Nachname", "Vorname", "E-Mail", "Abholdatum", "Gesamtsumme (€)", "Kommentar"];
      
      if (data.items && data.items.length > 0) {
        data.items.forEach(function(item) { headers.push(item.name); });
      }
      headers.push("Fleischbedarf (kg)");
      
      sheet.appendRow(headers);
      var headerRange = sheet.getRange(1, 1, 1, headers.length);
      headerRange.setFontWeight("bold");
      headerRange.setBackground("#8b0000");
      headerRange.setFontColor("#ffffff");
      sheet.setFrozenRows(1);
    }
    
    var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    var newRow = new Array(headers.length).fill("");
    
    newRow[0] = new Date().toLocaleString("de-DE");
    newRow[1] = data.nachname || "";
    newRow[2] = data.vorname || "";
    newRow[3] = data.email || "";
    newRow[4] = data.abholDatum || "";
    newRow[5] = data.totalSum ? data.totalSum.toFixed(2).replace('.', ',') + " €" : "0,00 €";
    newRow[6] = data.kommentar || "";
    
    var totalGrammOrder = 0;
    
    if (data.items && data.items.length > 0) {
      data.items.forEach(function(item) {
        var colIndex = headers.indexOf(item.name);
        if (colIndex !== -1) { newRow[colIndex] = item.displayQty; }
        if (item.totalWeightGramm) { totalGrammOrder += item.totalWeightGramm; }
      });
    }
    
    var weightColIndex = headers.indexOf("Fleischbedarf (kg)");
    if (weightColIndex !== -1) {
      newRow[weightColIndex] = (totalGrammOrder / 1000).toFixed(2).replace('.', ',') + " kg";
    }
    
    var lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      var lastRowFirstCell = sheet.getRange(lastRow, 1).getValue();
      if (lastRowFirstCell.toString().indexOf("GESAMTSUMME") !== -1) {
        sheet.deleteRow(lastRow);
      }
    }
    
    sheet.appendRow(newRow);
    
    var newLastRow = sheet.getLastRow();
    var sumRow = new Array(headers.length).fill("");
    sumRow[0] = "GESAMTSUMME";
    sumRow[5] = "=SUMME(F2:F" + newLastRow + ")";
    
    sheet.appendRow(sumRow);
    var sumRange = sheet.getRange(sheet.getLastRow(), 1, 1, headers.length);
    sumRange.setFontWeight("bold");
    sumRange.setBackground("#e8f5e9");
    
    return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"result": "error", "message": err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
